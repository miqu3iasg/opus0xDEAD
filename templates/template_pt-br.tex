% ============================================================
% TEMPLATE PARA ANOTAÇÕES DE ESTUDO EM LATEX (PORTUGUÊS)
% ============================================================
% Modelo de uso geral para converter anotações de estudo
% (originalmente escritas em Markdown ou em arquivos de código)
% para um documento LaTeX, para qualquer livro.
%
% Filosofia de design: corpo e capa em fonte padrão do LaTeX
% (Computer Modern), sem outros pacotes de fonte, sem cabeçalho,
% sem sumário. O corpo segue o modelo de um livro técnico clássico:
% texto simples, rodapé apenas com o número de página. A capa segue
% o modelo de uma folha de rosto simples de tese: papel branco,
% texto centralizado e um único desenho (um transistor) como
% única decoração.
%
% Esta é a versão em português (pt-BR). Todo o texto visível do
% documento está em português. A versão em inglês é o arquivo
% template_en.tex.
%
% Instruções de conversão:
%
% - Ao converter anotações de Markdown para este arquivo, mantenha o
%   idioma original das anotações exatamente como escrito.
% - Não altere, reescreva ou traduza o conteúdo das anotações.
%   Apenas formate adequadamente para LaTeX (seções, blocos de
%   código, ênfases, listas etc.).
% - Comentários de código fazem parte das anotações e também devem
%   ser mantidos no idioma original.
% - Se as anotações do livro estiverem em inglês, use a versão em
%   inglês do template (template_en.tex), para que todo o texto
%   visível (capa, créditos, títulos fixos) siga o idioma das notas.
%
% Uso:
%
% 1. Copie este arquivo para começar as anotações de um novo livro.
% 2. Preencha os campos marcados com <...> na seção "variáveis do
%    documento" abaixo. É o único lugar que você precisa editar
%    para configurar todo o documento.
% 3. Se o livro não for sobre programação, o pacote listings e o
%    bloco de código podem ser removidos.
% 4. Duplique o bloco de seção para cada capítulo de anotações.
% 5. Compile com pdflatex (duas vezes, para as referências cruzadas
%    ficarem corretas).

\documentclass[12pt,a4paper]{article}

% ============================================================
% Idioma e codificação
% ============================================================
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage[brazilian]{babel}
\usepackage{lmodern} % Latin Modern: a variante vetorial padrão do Computer Modern

% ============================================================
% Página
% ============================================================
\usepackage{geometry}
\geometry{margin=2.5cm}
\usepackage[protrusion=true,expansion=true]{microtype}

% ============================================================
% Recursos gráficos e fonte usados apenas na capa. A capa usa a
% mesma fonte do corpo (Computer Modern), como em uma folha de rosto
% clássica de tese. Para usar outra fonte apenas na capa, altere o
% comando abaixo (por exemplo, \fontfamily{bch}\selectfont para
% Charter, "ppl" para Palatino ou "ptm" para Times).
% ============================================================
\usepackage{tikz}
\usetikzlibrary{arrows.meta}
\newcommand{\coverserif}{\rmfamily}
\usepackage{xcolor}
% Esquema clássico: tinta preta sobre papel branco, cinza para texto secundário.
\definecolor{coverfg}{HTML}{000000}    % preto (traços e texto principal)
\definecolor{coverdim}{HTML}{555555}   % cinza escuro (texto secundário)

% ============================================================
% Listas
% ============================================================
\usepackage{enumitem}
\setlist{topsep=4pt,itemsep=2pt}

% ============================================================
% Rodapé: sem cabeçalho, apenas o número da página.
% ============================================================
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}
\fancyfoot[C]{\footnotesize\thepage}

% ============================================================
% Blocos de código: monoespaçado simples, sem moldura, sem fundo,
% sem numeração de linha.
% ============================================================
\usepackage{listings}

% Troque "Scheme" pela linguagem relevante para este livro, ou remova
% esta definição e use \lstset{language=...} com uma linguagem já
% suportada pelo listings (Python, C, Java etc.)
\lstdefinelanguage{Scheme}{
	morekeywords={},
	sensitive=true,
	morecomment=[l]{;},
}

\lstset{
	language=Scheme,
	basicstyle=\ttfamily\small,
	commentstyle=\ttfamily,
	frame=none,
	numbers=none,
	breaklines=true,
	showstringspaces=false,
	columns=fullflexible,
	xleftmargin=1.5em,
	% permite acentos e cedilha em português dentro dos blocos de
	% código (comentários de código, por exemplo). Se as notas forem
	% em inglês, este bloco pode ser removido.
	literate=
		{á}{{\'a}}1 {à}{{\`a}}1 {ã}{{\~a}}1 {â}{{\^a}}1
		{é}{{\'e}}1 {ê}{{\^e}}1 {í}{{\'i}}1
		{ó}{{\'o}}1 {ô}{{\^o}}1 {õ}{{\~o}}1 {ú}{{\'u}}1
		{ç}{{\c c}}1
		{Á}{{\'A}}1 {À}{{\`A}}1 {Ã}{{\~A}}1 {Â}{{\^A}}1
		{É}{{\'E}}1 {Ê}{{\^E}}1 {Í}{{\'I}}1
		{Ó}{{\'O}}1 {Ô}{{\^O}}1 {Õ}{{\~O}}1 {Ú}{{\'U}}1
		{Ç}{{\c C}}1,
}

% Termos/definições em linha: termo em negrito seguido do texto
% corrido, sem caixas e sem cor. Uso: \termo{Definição} texto...
\newcommand{\termo}[1]{\textbf{#1.}}

% ============================================================
% Variáveis do documento: preencha estas para cada novo livro
% ============================================================

\newcommand{\bookAuthor}{<AUTOR(ES) DO LIVRO>}
\newcommand{\studentName}{Miquéias Alves Medeiros}

% ============================================================
% Hyperlinks e metadados do PDF: links funcionam, mas ficam

\usepackage[hidelinks]{hyperref}

	pdftitle={\bookTitle: Anotações de estudo},

	pdfsubject={Anotações de estudo sobre \bookTitle},

% ============================================================
% Capa: folha de rosto simples, no estilo de tese. Papel branco,
% tudo centralizado, margens normais de página. No topo: tipo de
% documento e título. No meio: um desenho de um transistor bipolar
% abstrato (base, coletor e emissor) em traços finos, envolto por
% anéis e ligado a trilhas de circuito, única decoração. No rodapé:
% autor das notas e data.
% ============================================================
\begin{titlepage}
	\thispagestyle{empty}
	\centering
	\coverserif
	\vspace*{1.5cm}

	{\small\bfseries\color{coverdim}\MakeUppercase{Notas de estudo}\par}
	\vspace{0.4cm}
	{\color{coverfg}\rule{2cm}{0.6pt}\par}
	\vspace{1.0cm}

	{\bfseries\fontsize{24}{30}\selectfont\color{coverfg} \bookTitle\par}
	\vspace{0.8cm}
	{\normalsize\color{coverdim} Baseado na obra de \bookAuthor\par}

	\vfill

	% desenho do transistor: anéis concêntricos, envoltório, base/coletor/emissor,
	% trilhas paralelas e almofadas quadradas nas pontas das trilhas
	\begin{tikzpicture}[scale=0.65, coverfg]
		% anéis concêntricos discretos
		\foreach \r/\o in {5.6/0.22,7.0/0.12} {
				\draw[draw opacity=\o, line width=0.5pt] (0,0) circle (\r);
			}

		% envoltório do transistor
		\draw[line width=1.1pt] (0,0) circle (4.2);

		% base: barra do canal e terminal à esquerda
		\draw[line width=4pt] (-1.3,-2.1) -- (-1.3,2.1);
		\draw[line width=1.1pt] (-7.8,0) -- (-1.3,0);
		\fill (-7.8,0) circle (0.12);

		% coletor: diagonal e trilha que sobe e sai pela direita
		\draw[line width=1.1pt] (-1.3,1.0) -- (1.4,3.2) -- (1.4,6.6) -- (7.8,6.6);
		\fill (1.4,6.6) circle (0.12);

		% emissor: diagonal com seta e trilha que desce e sai pela direita
		\draw[line width=1.1pt, -{Stealth[length=3mm,width=2.4mm]}]
		(-1.3,-1.0) -- (1.4,-3.2);
		\draw[line width=1.1pt] (1.4,-3.2) -- (1.4,-4.6) -- (7.8,-4.6);
		\fill (1.4,-4.6) circle (0.12);

		% trilhas paralelas finas, ecoando a rota do coletor e do emissor
		\foreach \k/\o in {1/0.35,2/0.22,3/0.12} {
				\draw[draw opacity=\o, line width=0.5pt]
				(1.4+\k*0.35,3.2+\k*0.1) -- (1.4+\k*0.35,6.6+\k*0.35) -- (7.8,6.6+\k*0.35);
				\draw[draw opacity=\o, line width=0.5pt]
				(1.4+\k*0.35,-3.2-\k*0.1) -- (1.4+\k*0.35,-4.6-\k*0.35) -- (7.8,-4.6-\k*0.35);
			}

		% almofadas quadradas nas pontas das trilhas
		\foreach \yy in {6.6,-4.6} {
				\draw[line width=0.8pt] (6.7,\yy-0.3) rectangle ++(0.6,0.6);
			}
	\end{tikzpicture}

	\vfill

	{\large\bfseries\color{coverfg} \studentName\par}
	\vspace{0.3cm}
	{\small\color{coverdim} \today\par}
	\vspace*{1cm}
\end{titlepage}
