% ============================================================
% TEMPLATE PARA ANOTAÇÕES DE ESTUDO EM LATEX (PORTUGUÊS)
% ============================================================
% Modelo de uso geral para converter anotações de estudo
% (originalmente escritas em Markdown ou em arquivos de código)
% para um documento LaTeX, para qualquer livro.
%
% Filosofia de design: corpo em fonte padrão do LaTeX (Computer
% Modern), capa em Charter (serifada, do PSNFSS), sem outros pacotes
% de fonte, sem cabeçalho, sem sumário. O corpo segue o modelo de um
% livro técnico clássico: texto simples, rodapé apenas com o número
% de página. A personalidade fica concentrada na capa, sólida, de
% página inteira.
%
% Esta é a versão em português (pt-BR). Todo o texto visível do
% documento está em português. A versão em inglês é o arquivo
% template_en.tex.
%
% Instruções de conversão:
%
% - Ao converter anotações de Markdown para este arquivo, mantenha o
%   idioma original das anotações exatamente como escrito.
% - Não altere, reescreva ou traduza o conteúdo das anotações.
%   Apenas formate adequadamente para LaTeX (seções, blocos de
%   código, ênfases, listas etc.).
% - Comentários de código fazem parte das anotações e também devem
%   ser mantidos no idioma original.
% - Se as anotações do livro estiverem em inglês, use a versão em
%   inglês do template (template_en.tex), para que todo o texto
%   visível (capa, créditos, títulos fixos) siga o idioma das notas.
%
% Uso:
%
% 1. Copie este arquivo para começar as anotações de um novo livro.
% 2. Preencha os campos marcados com <...> na seção "variáveis do
%    documento" abaixo. É o único lugar que você precisa editar
%    para configurar todo o documento.
% 3. Se o livro não for sobre programação, o pacote listings e o
%    bloco de código podem ser removidos.
% 4. Duplique o bloco de seção para cada capítulo de anotações.
% 5. Compile com pdflatex (duas vezes, para as referências cruzadas
%    ficarem corretas).

\documentclass[12pt,a4paper]{article}

% ============================================================
% Idioma e codificação
% ============================================================
\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage[brazilian]{babel}
\usepackage{lmodern} % Latin Modern: a variante vetorial padrão do Computer Modern

% ============================================================
% Página
% ============================================================
\usepackage{geometry}
\geometry{margin=2.5cm}
\usepackage[protrusion=true,expansion=true]{microtype}

% ============================================================
% Recursos gráficos e fonte usados apenas na capa. A fonte da capa é
% a Charter (família "bch", do PSNFSS, incluída em qualquer instalação
% LaTeX): serifada, sólida e sóbria. O corpo do documento continua em
% Computer Modern. Para trocar a fonte da capa, altere apenas o código
% da família abaixo (por exemplo, "ppl" para Palatino ou "ptm" para Times).
% ============================================================
\usepackage{tikz}
\usetikzlibrary{calc,arrows.meta}
\newcommand{\coverserif}{\fontfamily{bch}\selectfont}
\usepackage{xcolor}
% Esquema monocromático: preto grafite e tons de cinza sobre branco.
\definecolor{coverbg}{HTML}{0C0C0D}    % grafite quase preto
\definecolor{coverfg}{HTML}{F2F2F2}    % branco suave (traços e texto principal)
\definecolor{coverdim}{HTML}{8C8C90}   % cinza médio (texto secundário)

% ============================================================
% Listas
% ============================================================
\usepackage{enumitem}
\setlist{topsep=4pt,itemsep=2pt}

% ============================================================
% Rodapé: sem cabeçalho, apenas o número da página.
% ============================================================
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}
\fancyfoot[C]{\footnotesize\thepage}

% ============================================================
% Blocos de código: monoespaçado simples, sem moldura, sem fundo,
% sem numeração de linha.
% ============================================================
\usepackage{listings}

% Troque "Scheme" pela linguagem relevante para este livro, ou remova
% esta definição e use \lstset{language=...} com uma linguagem já
% suportada pelo listings (Python, C, Java etc.)
\lstdefinelanguage{Scheme}{
	morekeywords={},
	sensitive=true,
	morecomment=[l]{;},
}

\lstset{
	language=Scheme,
	basicstyle=\ttfamily\small,
	commentstyle=\itshape,
	frame=none,
	numbers=none,
	breaklines=true,
	showstringspaces=false,
	columns=fullflexible,
	xleftmargin=1.5em,
	% permite acentos e cedilha em português dentro dos blocos de
	% código (comentários de código, por exemplo). Se as notas forem
	% em inglês, este bloco pode ser removido.
	literate=
		{á}{{\'a}}1 {à}{{\`a}}1 {ã}{{\~a}}1 {â}{{\^a}}1
		{é}{{\'e}}1 {ê}{{\^e}}1 {í}{{\'i}}1
		{ó}{{\'o}}1 {ô}{{\^o}}1 {õ}{{\~o}}1 {ú}{{\'u}}1
		{ç}{{\c c}}1
		{Á}{{\'A}}1 {À}{{\`A}}1 {Ã}{{\~A}}1 {Â}{{\^A}}1
		{É}{{\'E}}1 {Ê}{{\^E}}1 {Í}{{\'I}}1
		{Ó}{{\'O}}1 {Ô}{{\^O}}1 {Õ}{{\~O}}1 {Ú}{{\'U}}1
		{Ç}{{\c C}}1,
}

% Termos/definições em linha: termo em negrito seguido do texto
% corrido, sem caixas e sem cor. Uso: \termo{Definição} texto...
\newcommand{\termo}[1]{\textbf{#1.}}

% ============================================================
% Variáveis do documento: preencha estas para cada novo livro

\newcommand{\studentName}{Miquéias Alves Medeiros}

% ============================================================
% Hyperlinks e metadados do PDF: links funcionam, mas ficam

% ============================================================
% Capa: página inteira, sangria total, monocromática. Fundo grafite
% e um único motivo gráfico: um transistor bipolar abstrato (base,
% coletor e emissor) desenhado em traços finos, envolto por anéis
% concêntricos e ligado a trilhas de circuito que saem da página.
% Tipografia serifada (Charter) em caixa-alta. Sem cor além de
% branco e cinza.
% ============================================================
\begin{titlepage}
	\newgeometry{margin=0cm}
	\thispagestyle{empty}
	\begin{tikzpicture}[remember picture, overlay]
		\def\cx{13.2}
		\def\cy{10.6}
		% fundo sólido, sangria total
		\fill[coverbg] (current page.south west) rectangle (current page.north east);

		\begin{scope}[shift={(current page.south west)}]
			\clip (0,0) rectangle (21,29.7);

			% anéis concêntricos discretos (sangram pela borda)
			\foreach \r/\o in {5.6/0.22,7.0/0.14,8.4/0.08,9.8/0.04} {
					\draw[coverfg, draw opacity=\o, line width=0.5pt] (\cx,\cy) circle (\r cm);
				}

			% envoltório do transistor
			\draw[coverfg, line width=1.1pt] (\cx,\cy) circle (4.2cm);

			% base: barra do canal e terminal à esquerda
			\draw[coverfg, line width=4pt] (11.9,\cy-2.1) -- (11.9,\cy+2.1);
			\draw[coverfg, line width=1.1pt] (2.6,\cy) -- (11.9,\cy);
			\fill[coverfg] (2.6,\cy) circle (0.12cm);

			% coletor: diagonal e trilha que sobe e sai pela direita
			\draw[coverfg, line width=1.1pt] (11.9,\cy+1.0) -- (14.6,\cy+3.2)
			-- (14.6,17.2) -- (21,17.2);
			\fill[coverfg] (14.6,17.2) circle (0.12cm);

			% emissor: diagonal com seta e trilha que desce e sai pela direita
			\draw[coverfg, line width=1.1pt, -{Stealth[length=3.2mm,width=2.6mm]}]
			(11.9,\cy-1.0) -- (14.6,\cy-3.2);
			\draw[coverfg, line width=1.1pt] (14.6,\cy-3.2) -- (14.6,6.0) -- (21,6.0);
			\fill[coverfg] (14.6,6.0) circle (0.12cm);

			% trilhas paralelas finas, ecoando a rota do coletor e do emissor
			\foreach \k/\o in {1/0.35,2/0.22,3/0.12} {
					\draw[coverfg, draw opacity=\o, line width=0.5pt]
					(14.6+\k*0.35,\cy+3.2+\k*0.1) -- (14.6+\k*0.35,17.2+\k*0.35) -- (21,17.2+\k*0.35);
					\draw[coverfg, draw opacity=\o, line width=0.5pt]
					(14.6+\k*0.35,\cy-3.2-\k*0.1) -- (14.6+\k*0.35,6.0-\k*0.35) -- (21,6.0-\k*0.35);
				}

			% almofadas quadradas nas pontas das trilhas
			\foreach \yy in {17.2,6.0} {
					\draw[coverfg, line width=0.8pt] (19.9,\yy-0.3) rectangle ++(0.6,0.6);
				}
		\end{scope}

		% pré-título
		\node[anchor=north west, text=coverdim, font=\coverserif\small\bfseries,
			xshift=2.6cm, yshift=-3.0cm]
		at (current page.north west) {\MakeUppercase{Notas de estudo}};
		\draw[coverfg, line width=1.1pt]
		($(current page.north west)+(2.6cm,-3.7cm)$) --
		($(current page.north west)+(4.6cm,-3.7cm)$);

		% título: bloco alinhado à esquerda
		\node[anchor=north west, text=coverfg, font=\coverserif\bfseries\fontsize{27}{33}\selectfont,
			execute at begin node={\hyphenpenalty=10000\relax\exhyphenpenalty=10000\relax},
			align=left, text width=12.5cm, xshift=2.6cm, yshift=-6.2cm]
		at (current page.north west) {\MakeUppercase{\bookTitle}};

		% subtítulo (autor da obra)
		\node[anchor=south west, text=coverdim, font=\coverserif\small,
			execute at begin node={\hyphenpenalty=10000\relax\exhyphenpenalty=10000\relax},
			align=left, text width=8cm, xshift=2.6cm, yshift=3.6cm]
		at (current page.south west) {Baseado na obra de \bookAuthor};

		% autor das notas e data, rodapé da capa
		\node[anchor=south west, text=coverfg, font=\coverserif\bfseries,
			xshift=2.6cm, yshift=2.1cm]
		at (current page.south west) {\studentName};
		\node[anchor=south west, text=coverdim, font=\coverserif\footnotesize,
			xshift=2.6cm, yshift=1.4cm]
		at (current page.south west) {\today};
	\end{tikzpicture}
	\restoregeometry
\end{titlepage}
